```sql
-- DuckDB: no download, no credentials
ATTACH 'https://omicidx.cancerdatasci.org/data/latest/omicidx.duckdb' AS omicidx (READ_ONLY);
SELECT run_accession, study_title FROM omicidx.sradb.rnaseq_runs WHERE taxon_id = 9606 LIMIT 10;
```
